# Google STT/TTS (Flutter speech_to_text + flutter_tts).

# Kokoro (English): Download model + voices from https://huggingface.co/hexgrad/Kokoro-82M
# Meta MMS-TTS (1100+ languages including Tagalog facebook/mms-tts-tag and English facebook/mms-tts-eng):

onnxruntime>=1.20.1

transformers>=4.38.0

# CPU/GPU helpers — optional but speeds up STT.
